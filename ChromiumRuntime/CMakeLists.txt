cmake_minimum_required(VERSION 3.21)
project(RadiusChromium LANGUAGES C CXX OBJCXX)
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
# The package builder signs the completed development binaries explicitly.
# Xcode rejects CEF's empty signing identity for a dynamic-library target.
set(CMAKE_XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
if(NOT OS_MAC OR NOT USE_SANDBOX)
  message(FATAL_ERROR "This adapter requires macOS and the CEF sandbox")
endif()
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
add_library(RadiusChromiumBridge SHARED RadiusChromiumBridge.mm)
list(REMOVE_ITEM CEF_LINKER_FLAGS "-Wl,-pie")
SET_LIBRARY_TARGET_PROPERTIES(RadiusChromiumBridge)
target_compile_options(RadiusChromiumBridge PRIVATE ${CEF_COMPILER_FLAGS} ${CEF_CXX_COMPILER_FLAGS})
target_include_directories(RadiusChromiumBridge PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../Sources/RadiusEngineABI/include")
target_link_libraries(RadiusChromiumBridge libcef_dll_wrapper ${CEF_STANDARD_LIBS})
set_target_properties(RadiusChromiumBridge PROPERTIES PREFIX "" CXX_STANDARD 20 OBJCXX_STANDARD 20)
foreach(suffix_list ${CEF_HELPER_APP_SUFFIXES})
  string(REPLACE ":" ";" parts "${suffix_list}")
  list(GET parts 0 name_suffix)
  list(GET parts 1 target_suffix)
  list(GET parts 2 id_suffix)
  set(helper_target "RadiusChromiumHelper${target_suffix}")
  set(helper_name "RadiusChromium Helper${name_suffix}")
  set(helper_id "org.radius.browser.chromium.helper${id_suffix}")
  configure_file(Helper-Info.plist.in "${CMAKE_CURRENT_BINARY_DIR}/${helper_target}.plist" @ONLY)
  add_executable(${helper_target} MACOSX_BUNDLE Helper.cc)
  SET_EXECUTABLE_TARGET_PROPERTIES(${helper_target})
  target_link_libraries(${helper_target} libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  set_target_properties(${helper_target} PROPERTIES OUTPUT_NAME "${helper_name}"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${helper_id}"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/${helper_target}.plist")
  add_dependencies(RadiusChromiumBridge ${helper_target})
endforeach()
